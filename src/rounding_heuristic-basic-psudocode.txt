INPUT:
  - MIPProblem (A, b, c, bounds, integrality)
  - fractional solution x_frac
  - number of attempts K

OUTPUT:
  - best feasible integer solution found

ALGORITHM:
  best_solution ← infeasible
  [ 
  	best_solution.feasible ← false
  	best_solution.objective ← +∞
  ]

  parallel for k = 1 to K:
      x_int ← round x_frac using strategy k
      if x_int violates bounds:
          continue

      if x_int violates constraints:
          continue

      obj ← compute objective(x_int)

      if x_int is better than local best:
          update local best

  reduce all local best solutions into global best

  return global best
